cmake_minimum_required(VERSION 3.16)
project(qutaq-launcher VERSION 0.1 LANGUAGES C CXX)  # C is for the code generated by wayland-scanner

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

# Qt >= 6.5: per-window layer-shell integration (the launcher is a layer surface, its menus are
# ordinary xdg_popups) and QNativeInterface::QWaylandApplication::seat() for activating windows.
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets DBus WaylandClient)
# UI translations (lrelease). Without Qt Linguist the launcher still builds, in English only.
find_package(Qt6 6.5 QUIET OPTIONAL_COMPONENTS LinguistTools)

# LayerShellQt 5.x (Plasma 5) turns ALL windows into layer surfaces, including QMenu,
# so menus open full screen. Hence 6.x only.
find_package(LayerShellQt 6.0 REQUIRED)

qt_add_executable(qutaq-launcher
    src/main.cpp
    src/desktopentries.h    src/desktopentries.cpp
    src/launchermodel.h     src/launchermodel.cpp
    src/launcherview.h      src/launcherview.cpp
    src/appmenu.h           src/appmenu.cpp
    src/trash.h             src/trash.cpp
    src/devices.h           src/devices.cpp
    src/toplevelmanager.h   src/toplevelmanager.cpp
    src/icons.h             src/icons.cpp
    src/settings.h
)

# Generates qwayland-*.h / wayland-*-client-protocol.h from the protocol XML
qt6_generate_wayland_protocol_client_sources(qutaq-launcher
    FILES ${CMAKE_CURRENT_SOURCE_DIR}/protocols/wlr-foreign-toplevel-management-unstable-v1.xml
)

# translations/qutaq-launcher_<lang>.ts → .qm files built into the program (:/i18n).
file(GLOB QUTAQ_TS_FILES ${CMAKE_CURRENT_SOURCE_DIR}/translations/qutaq-launcher_*.ts)
if(TARGET Qt6::lrelease AND QUTAQ_TS_FILES)
    qt_add_translations(qutaq-launcher
        TS_FILES ${QUTAQ_TS_FILES}
        RESOURCE_PREFIX /i18n
        LUPDATE_OPTIONS -no-obsolete -locations none
    )
else()
    message(WARNING "Qt6 LinguistTools not found: building without translations (English only)")
endif()

target_link_libraries(qutaq-launcher PRIVATE
    Qt6::Core Qt6::Gui Qt6::Widgets Qt6::DBus Qt6::WaylandClient
    LayerShellQt::Interface
)
target_compile_definitions(qutaq-launcher PRIVATE
    LSQ_VERSION_MAJOR=${LayerShellQt_VERSION_MAJOR}
    LSQ_VERSION_MINOR=${LayerShellQt_VERSION_MINOR}
)

install(TARGETS qutaq-launcher RUNTIME DESTINATION bin)
